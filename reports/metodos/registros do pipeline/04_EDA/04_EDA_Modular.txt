código:


"""
=============================================================================
EDA PASSO 4: O FATOR HUMANO INTRAESCOLAR (SOBRECARGA & REGULARIDADE DOCENTE)
=============================================================================
Objetivo Pedagógico e Científico:
  No Passo 3, descobrimos que após o INSE, os maiores preditores da nota de
  Matemática são o Esforço Docente (IED, relação negativa r = -0.23) e a
  Regularidade do Vínculo Docente (IRD, relação positiva r = +0.17).

  Neste passo, aprofundamos o diagnóstico do corpo docente:
  1. Análise bivariada detalhada de IED (esforço) e IRD (estabilidade);
  2. Matriz dos 4 Quadrantes Docentes da Rede Paulista:
     - Q1: Alta Sobrecarga & Baixa Regularidade (Cenário mais crítico);
     - Q2: Alta Sobrecarga & Alta Regularidade;
     - Q3: Baixa Sobrecarga & Baixa Regularidade;
     - Q4: Baixa Sobrecarga & Alta Regularidade (Cenário ideal);
  3. Gap Pedagógico: Qual a diferença de nota entre os quadrantes extremos?
  4. Teste de Robustez: O efeito docente sobrevive após controlar pelo INSE?
  5. Geração de Gráficos Científicos em 300 DPI:
     - Dispersão IED vs. Target colorida por IRD;
     - Boxplot comparativo dos 4 Quadrantes Docentes.

Saída:
  - reports/figures/eda_04_fatores_docentes_ied_ird.png
=============================================================================
"""

from pathlib import Path
import pandas as pd
import numpy as np

# Configurar caminhos do projeto
PROJECT_ROOT = Path(__file__).resolve().parents[2]
GOLD_DIR = PROJECT_ROOT / "data" / "gold"
ARQUIVO_GOLD = GOLD_DIR / "tcc_dataset_analitico_final.parquet"
FIGURES_DIR = PROJECT_ROOT / "reports" / "figures"
FIGURES_DIR.mkdir(parents=True, exist_ok=True)

def analisar_fatores_docentes():
    print("=" * 75)
    print("EDA PASSO 4: O FATOR HUMANO INTRAESCOLAR (DOCENTES: IED vs. IRD)")
    print("=" * 75)

    # 1. Carregamento e Seleção de Dados
    print("\n1. Carregando dados da Camada Gold...")
    df = pd.read_parquet(ARQUIVO_GOLD)
    
    cols_docentes = ['CODESC', 'NOMESC', 'MUN', 'MEDIA_INSE', 'IED_SCORE_MEDIO', 'IED_ESFORCO_ALTO', 'IRD_MEDIO', 'TARGET_TRIENAL_MAT']
    df_doc = df[cols_docentes].dropna().copy()
    n = len(df_doc)
    print(f"   -> Escolas com indicadores docentes completos: {n:,} de {len(df):,} ({n/len(df)*100:.1f}%)")

    # 2. Distribuição das Variáveis Docentes
    print("\n2. Estatísticas do Corpo Docente na Rede Estadual:")
    print(f"   - Esforço Docente Médio (IED):   Média = {df_doc['IED_SCORE_MEDIO'].mean():.2f} | Mediana = {df_doc['IED_SCORE_MEDIO'].median():.2f} (Escala 1 a 6)")
    print(f"   - Sobrecarga Severa (Cat 5 e 6): Média = {df_doc['IED_ESFORCO_ALTO'].mean():.2f}% da equipe em regime de alta sobrecarga")
    print(f"   - Regularidade Docente (IRD):    Média = {df_doc['IRD_MEDIO'].mean():.2f} | Mediana = {df_doc['IRD_MEDIO'].median():.2f} (Escala 0 a 5)")

    # 3. Construção dos 4 Quadrantes Docentes da Rede Paulista
    # Usando a mediana da rede como ponto de corte para neutralidade
    mediana_ied = df_doc['IED_SCORE_MEDIO'].median()
    mediana_ird = df_doc['IRD_MEDIO'].median()

    def classificar_quadrante(row):
        ied_alto = row['IED_SCORE_MEDIO'] > mediana_ied
        ird_alto = row['IRD_MEDIO'] > mediana_ird
        if ied_alto and not ird_alto:
            return "Q1: Alta Sobrecarga / Baixa Regularidade (Crítico)"
        elif ied_alto and ird_alto:
            return "Q2: Alta Sobrecarga / Alta Regularidade"
        elif not ied_alto and not ird_alto:
            return "Q3: Baixa Sobrecarga / Baixa Regularidade"
        else:
            return "Q4: Baixa Sobrecarga / Alta Regularidade (Ideal)"

    df_doc['QUADRANTE_DOCENTE'] = df_doc.apply(classificar_quadrante, axis=1)

    print("\n3. Distribuição das Escolas pelos 4 Quadrantes Docentes:")
    ordem_quadrantes = [
        "Q1: Alta Sobrecarga / Baixa Regularidade (Crítico)",
        "Q2: Alta Sobrecarga / Alta Regularidade",
        "Q3: Baixa Sobrecarga / Baixa Regularidade",
        "Q4: Baixa Sobrecarga / Alta Regularidade (Ideal)"
    ]
    
    resumo_quadrantes = df_doc.groupby('QUADRANTE_DOCENTE', observed=True).agg(
        Qtd_Escolas=('TARGET_TRIENAL_MAT', 'count'),
        IED_Medio=('IED_SCORE_MEDIO', 'mean'),
        IRD_Medio=('IRD_MEDIO', 'mean'),
        INSE_Medio=('MEDIA_INSE', 'mean'),
        Nota_Media=('TARGET_TRIENAL_MAT', 'mean'),
        Nota_Mediana=('TARGET_TRIENAL_MAT', 'median'),
        Nota_Min=('TARGET_TRIENAL_MAT', 'min'),
        Nota_Max=('TARGET_TRIENAL_MAT', 'max')
    ).reindex(ordem_quadrantes)

    for q, row in resumo_quadrantes.iterrows():
        print(f"\n   [{q}]")
        print(f"     * Volume de Escolas: {int(row['Qtd_Escolas']):>4} ({row['Qtd_Escolas']/n*100:.1f}% da rede)")
        print(f"     * IED Médio: {row['IED_Medio']:.2f} | IRD Médio: {row['IRD_Medio']:.2f} | INSE Médio: {row['INSE_Medio']:.2f}")
        print(f"     * Nota Média de Matemática: {row['Nota_Media']:.2f}% (Mediana: {row['Nota_Mediana']:.2f}%)")
        print(f"     * Amplitude das Notas:      [{row['Nota_Min']:.2f}%, {row['Nota_Max']:.2f}%]")

    gap_docente = resumo_quadrantes.loc['Q4: Baixa Sobrecarga / Alta Regularidade (Ideal)', 'Nota_Media'] - \
                  resumo_quadrantes.loc['Q1: Alta Sobrecarga / Baixa Regularidade (Crítico)', 'Nota_Media']
    print("\n" + "-" * 65)
    print(f"O 'GAP DO CORPO DOCENTE' NA REDE ESTADUAL:")
    print(f"Escolas no ambiente docente ideal (baixa sobrecarga e alta estabilidade)")
    print(f"apresentam nota média {gap_docente:+.2f} pontos percentuais superior")
    print(f"às escolas no ambiente crítico (alta sobrecarga e alta rotatividade).")
    print("-" * 65)

    # 4. Teste Econométrico: O Efeito Docente Sobrevive ao Controle do INSE?
    print("\n4. Teste de Regressão Múltipla: Isolando o Efeito Docente do INSE:")
    X_mat = np.column_stack([
        np.ones(len(df_doc)),
        df_doc['MEDIA_INSE'].values,
        df_doc['IED_SCORE_MEDIO'].values,
        df_doc['IRD_MEDIO'].values
    ])
    Y_vec = df_doc['TARGET_TRIENAL_MAT'].values
    betas, _, _, _ = np.linalg.lstsq(X_mat, Y_vec, rcond=None)
    y_pred_multi = X_mat @ betas
    r2_multi = 1 - (np.sum((Y_vec - y_pred_multi)**2) / np.sum((Y_vec - Y_vec.mean())**2))

    print(f"   Modelo: Target = b0 + (b1 * INSE) + (b2 * IED) + (b3 * IRD)")
    print(f"   - Coeficiente INSE (b1): {betas[1]:+.3f} p.p. por ponto de INSE")
    print(f"   - Coeficiente IED  (b2): {betas[2]:+.3f} p.p. por ponto de Esforço (Negativo!)")
    print(f"   - Coeficiente IRD  (b3): {betas[3]:+.3f} p.p. por ponto de Regularidade (Positivo!)")
    print(f"   - R² do Modelo Conjunto: {r2_multi*100:.2f}%")
    print(f"   -> Conclusão: MESMO CONTROLANDO PELO INSE, o esforço e a estabilidade docente")
    print(f"      continuam com impactos estatisticamente robustos e independentes na nota!")

    # 5. Geração de Gráfico Científico em Alta Resolução (300 DPI)
    print("\n5. Gerando painel gráfico científico em 300 DPI...")
    try:
        import matplotlib.pyplot as plt
        import seaborn as sns

        plt.style.use('seaborn-v0_8-whitegrid' if 'seaborn-v0_8-whitegrid' in plt.style.available else 'default')
        fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(15, 6), dpi=300)

        # Painel 1: Dispersão IED vs. Target com gradiente por IRD
        sc = ax1.scatter(
            df_doc['IED_SCORE_MEDIO'], df_doc['TARGET_TRIENAL_MAT'],
            c=df_doc['IRD_MEDIO'], cmap='viridis', alpha=0.45, s=25, edgecolors='none'
        )
        cbar = fig.colorbar(sc, ax=ax1)
        cbar.set_label('Regularidade Docente (IRD: 0 a 5)', fontsize=10)
        
        # Reta de tendência do IED
        b1_ied, b0_ied = np.polyfit(df_doc['IED_SCORE_MEDIO'], df_doc['TARGET_TRIENAL_MAT'], deg=1)
        x_ied = np.linspace(df_doc['IED_SCORE_MEDIO'].min(), df_doc['IED_SCORE_MEDIO'].max(), 100)
        ax1.plot(x_ied, b0_ied + b1_ied * x_ied, color='#d62728', linewidth=2.2, label=f'Tendência: b = {b1_ied:.2f} p.p./IED')
        
        ax1.set_title('A: Esforço Docente (IED) vs. Desempenho em Matemática', fontsize=11.5, fontweight='bold', pad=10)
        ax1.set_xlabel('Score de Esforço Docente (IED: 1 a 6 | Maior = Mais Sobrecarga)', fontsize=10)
        ax1.set_ylabel('Média Trienal em Matemática (% de acertos)', fontsize=10)
        ax1.legend(loc='upper right', frameon=True, facecolor='white')

        # Painel 2: Boxplot dos 4 Quadrantes Docentes
        cores_quadrantes = ['#e41a1c', '#ff7f00', '#377eb8', '#4daf4a']
        sns.boxplot(
            data=df_doc, x='QUADRANTE_DOCENTE', y='TARGET_TRIENAL_MAT',
            order=ordem_quadrantes, palette=cores_quadrantes, ax=ax2, width=0.55,
            fliersize=3, flierprops={'alpha': 0.4}
        )
        ax2.set_title('B: Comparação das Notas pelos 4 Quadrantes Docentes', fontsize=11.5, fontweight='bold', pad=10)
        ax2.set_xlabel('')
        ax2.set_ylabel('Média Trienal em Matemática (% de acertos)', fontsize=10)
        ax2.set_xticklabels(['Q1\n(Crítico)', 'Q2\n(Sobrecarga)', 'Q3\n(Rotatividade)', 'Q4\n(Ideal)'], fontsize=9.5)

        # Adicionar médias textuais nos boxplots
        for i, q in enumerate(ordem_quadrantes):
            media_q = resumo_quadrantes.loc[q, 'Nota_Media']
            ax2.text(i, media_q + 0.6, f'μ = {media_q:.1f}%', ha='center', fontsize=9, fontweight='bold', color='#111111')

        fig.tight_layout()
        caminho_fig = FIGURES_DIR / "eda_04_fatores_docentes_ied_ird.png"
        fig.savefig(caminho_fig)
        plt.close(fig)
        print(f"   [SUCESSO] Gráfico salvo com sucesso em:\n   {caminho_fig}")

    except Exception as e:
        print(f"   [AVISO] Erro ao renderizar gráfico: {e}")

    print("\n" + "=" * 75)
    print("PASSO 4 CONCLUÍDO COM SUCESSO!")
    print("=" * 75)

if __name__ == "__main__":
    analisar_fatores_docentes()
